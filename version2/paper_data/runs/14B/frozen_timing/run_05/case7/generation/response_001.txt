```python
#!/usr/bin/env python3
import sys
import math
import numpy as np
import matplotlib.pyplot as plt

def get_particle_mass(obj):
    """Return appropriate mass based on particle type."""
    if 'type' not in obj:
        return obj.get('mass', obj.get('jmass', 0.0))

    ptype = obj['type']
    if ptype == 0:    # photon
        return 0.0
    elif ptype == 1:  # electron
        return 0.0
    elif ptype == 2:  # muon
        return 0.0
    elif ptype == 3:  # tau
        return 1.777
    elif ptype == 4:  # jet
        return obj['jmass']
    else:
        return obj.get('mass', 0.0)

def four_momentum(obj):
    """Return (E, px, py, pz) from an object dict."""
    pt  = obj['pt']
    eta = obj['eta']
    phi = obj['phi']
    m   = get_particle_mass(obj)

    px = pt * math.cos(phi)
    py = pt * math.sin(phi)
    pz = pt * math.sinh(eta)
    E  = math.sqrt(px**2 + py**2 + pz**2 + m**2)
    return E, px, py, pz

def obj_from_four_momentum(E, px, py, pz):
    """
    Reconstruct a composite object dict from four-momentum (E, px, py, pz).
    Returns a dict with pt, eta, phi, and mass.
    """
    pt = math.sqrt(px**2 + py**2)
    phi = math.atan2(py, px)

    if pt > 0.0:
        eta = math.asinh(pz / pt)
    elif pz != 0.0:
        eta = math.copysign(float("inf"), pz)
    else:
        eta = 0.0

    p2 = px**2 + py**2 + pz**2
    m2 = E**2 - p2
    mass = math.sqrt(m2) if m2 > 0.0 else 0.0

    return {
        "pt": pt,
        "eta": eta,
        "phi": phi,
        "mass": mass
    }

def sum_objects(*args):
    """Return composite object from any combination of particles/lists."""
    particles = []
    for arg in args:
        if arg is None:
            continue
        if isinstance(arg, dict):
            particles.append(arg)
        elif isinstance(arg, list):
            particles.extend(p for p in arg if p is not None)

    if not particles:
        return None

    E_tot, px_tot, py_tot, pz_tot = 0.0, 0.0, 0.0, 0.0
    for p in particles:
        E, px, py, pz = four_momentum(p)
        E_tot += E
        px_tot += px
        py_tot += py
        pz_tot += pz

    return obj_from_four_momentum(E_tot, px_tot, py_tot, pz_tot)

def helper_invariant_mass(*args):
    """Return invariant mass from any combination of particles/lists."""
    obj = sum_objects(*args)
    return obj['mass'] if obj is not None else 0.0

def helper_delta_phi(phi1, phi2):
    """Compute Δφ normalized to [-π, π]."""
    dphi = phi1 - phi2
    while dphi > math.pi:
        dphi -= 2 * math.pi
    while dphi < -math.pi:
        dphi += 2 * math.pi
    return dphi

def helper_delta_r(obj1, obj2):
    """Compute ΔR between two objects."""
    deta = obj1['eta'] - obj2['eta']
    dphi = helper_delta_phi(obj1['phi'], obj2['phi'])
    return math.sqrt(deta**2 + dphi**2)

def helper_get_leading(particles):
    """Return leading (highest pT) particle or None."""
    if not particles:
        return None
    return max(particles, key=lambda p: p['pt'])

def helper_get_subleading(particles):
    """Return subleading (second highest pT) particle or None."""
    if len(particles) < 2:
        return None
    sorted_p = sorted(particles, key=lambda p: p['pt'], reverse=True)
    return sorted_p[1]

def get_taus(event):
    """Return all taus."""
    return [o for o in event['objects'] if o['type'] == 3]

def get_jets(event):
    """Return all jets."""
    return [o for o in event['objects'] if o['type'] == 4]

def get_bjets(event):
    """Return all b-tagged jets."""
    return [o for o in event['objects'] if o['type'] == 4 and int(o['btag']) == 1]

def n_taus(event, pt_cut=0.0, eta_cut=float('inf')):
    """Count taus passing pT and |η| cuts."""
    taus = get_taus(event)
    return sum(1 for t in taus if t['pt'] > pt_cut and abs(t['eta']) < eta_cut)

def n_jets(event, pt_cut=0.0, eta_cut=float('inf')):
    """Count jets passing pT and |η| cuts."""
    jets = get_jets(event)
    return sum(1 for j in jets if j['pt'] > pt_cut and abs(j['eta']) < eta_cut)

def opposite_sign(obj1, obj2):
    """Check if two objects have opposite charge."""
    return obj1['charge'] * obj2['charge'] < 0

def helper_delta_eta(obj1, obj2):
    """Compute Δη between two objects."""
    return obj1['eta'] - obj2['eta']

def centrality_between_jets(obj, jet1, jet2):
    """
    Compute eta centrality of obj relative to two jets.

    centrality = |eta_obj - eta_center_jj| / |delta_eta_jj|
    Smaller values mean obj is closer to the midpoint between the two jets.
    """
    eta_gap = abs(jet1["eta"] - jet2["eta"])
    if eta_gap <= 0.0:
        return 0.0

    eta_center = 0.5 * (jet1["eta"] + jet2["eta"])
    return abs(obj["eta"] - eta_center) / eta_gap

def read_lhco(filename):
    """Parse LHCO file and return list of events."""
    events = []
    current = None

    with open(filename) as f:
        for line in f:
            line = line.strip()
            if not line or line.startswith('#'):
                continue

            parts = line.split()
            if not parts:
                continue

            # New event header (index = 0)
            if parts[0] == '0':
                if len(parts) < 2:
                    continue
                try:
                    event_id = int(parts[1])
                except ValueError:
                    continue
                if current is not None:
                    events.append(current)
                current = {'id': event_id, 'objects': []}

            # Object line
            elif current is not None:
                if len(parts) < 7:
                    continue
                try:
                    obj = {
                        'type':   int(parts[1]),
                        'eta':    float(parts[2]),
                        'phi':    float(parts[3]),
                        'pt':     float(parts[4]),
                        'jmass':  float(parts[5]),
                        'ntrk':   float(parts[6]),
                        'btag':   float(parts[7]) if len(parts) > 7 else 0.0,
                        'had_em': float(parts[8]) if len(parts) > 8 else 0.0
                    }
                except ValueError:
                    continue

                obj['charge'] = 1 if obj['ntrk'] > 0 else (-1 if obj['ntrk'] < 0 else 0)
                current['objects'].append(obj)

    if current is not None:
        events.append(current)

    return events

def main():
    if len(sys.argv) < 2:
        print("Usage: python script.py <lhco_file1> [lhco_file2] ...")
        return

    files = sys.argv[1:]
    events = []
    for f in files:
        events.extend(read_lhco(f))

    cutflow = {
        "Initial events": len(events),
        "After tau selection": 0,
        "After jet selection": 0,
        "After 2 taus": 0,
        "After 2 jets": 0,
        "After opposite sign": 0,
        "After m_jj > 400 GeV": 0,
        "After |delta eta| > 2.5": 0,
        "After m_tt in [60, 150] GeV": 0,
        "After b-jet veto": 0,
        "Final selected events": 0
    }

    ditau_mass_list = []
    dijet_mass_list = []
    delta_eta_jj_list = []
    leading_tau_pt_list = []
    subleading_tau_pt_list = []
    leading_jet_pt_list = []
    subleading_jet_pt_list = []
    delta_r_tt_list = []
    delta_phi_tt_jj_list = []
    centrality_tt_jj_list = []

    for event in events:
        # Step 1: Select taus with pT > 20 GeV and |eta| < 2.3
        taus = [t for t in get_taus(event) if t['pt'] > 20 and abs(t['eta']) < 2.3]
        if len(taus) < 2:
            continue
        cutflow["After tau selection"] += 1

        # Step 2: Select jets with pT > 30 GeV and |eta| < 4.7
        jets = [j for j in get_jets(event) if j['pt'] > 30 and abs(j['eta']) < 4.7]
        if len(jets) < 2:
            continue
        cutflow["After jet selection"] += 1

        # Step 3: Require at least 2 taus
        if len(taus) < 2:
            continue
        cutflow["After 2 taus"] += 1

        # Step 4: Define leading and subleading taus
        leading_tau = helper_get_leading(taus)
        subleading_tau = helper_get_subleading(taus)

        # Step 5: Define ditau_obj
        ditau_obj = sum_objects(leading_tau, subleading_tau)

        # Step 6: Require at least 2 jets (VBF topology)
        if len(jets) < 2:
            continue
        cutflow["After 2 jets"] += 1

        # Step 7: Define leading and subleading jets
        leading_jet = helper_get_leading(jets)
        subleading_jet = helper_get_subleading(jets)

        # Step 8: Define dijet_obj
        dijet_obj = sum_objects(leading_jet, subleading_jet)

        # Step 9: Require opposite-sign tau pair
        if not opposite_sign(leading_tau, subleading_tau):
            continue
        cutflow["After opposite sign"] += 1

        # Step 10: Select invariant mass of the leading and subleading jets (m_jj) > 400 GeV
        m_jj = helper_invariant_mass(leading_jet, subleading_jet)
        if m_jj < 400:
            continue
        cutflow["After m_jj > 400 GeV"] += 1

        # Step 11: Select |delta eta| between the two leading jets > 2.5
        delta_eta_jj = abs(helper_delta_eta(leading_jet, subleading_jet))
        if delta_eta_jj < 2.5:
            continue
        cutflow["After |delta eta| > 2.5"] += 1

        # Step 12: Select ditau invariant mass between 60 and 150 GeV
        m_tt = helper_invariant_mass(leading_tau, subleading_tau)
        if not (60 < m_tt < 150):
            continue
        cutflow["After m_tt in [60, 150] GeV"] += 1

        # Step 13: Veto events with b-tagged jets among the selected jets
        if any(int(j['btag']) == 1 for j in jets):
            continue
        cutflow["After b-jet veto"] += 1

        # Step 14: Collect observables
        cutflow["Final selected events"] += 1

        ditau_mass_list.append(m_tt)
        dijet_mass_list.append(m_jj)
        delta_eta_jj_list.append(delta_eta_jj)
        leading_tau_pt_list.append(leading_tau['pt'])
        subleading_tau_pt_list.append(subleading_tau['pt'])
        leading_jet_pt_list.append(leading_jet['pt'])
        subleading_jet_pt_list.append(subleading_jet['pt'])
        delta_r_tt_list.append(helper_delta_r(leading_tau, subleading_tau))
        delta_phi_tt_jj_list.append(helper_delta_phi(ditau_obj['phi'], dijet_obj['phi']))
        centrality_tt_jj_list.append(centrality_between_jets(ditau_obj, leading_jet, subleading_jet))

    # Print cutflow
    for key, value in cutflow.items():
        print(f"{key}: {value}")

    # Plot histograms
    def plot_hist(data, bins, xlabel, filename):
        plt.figure()
        plt.hist(data, bins=bins, density=True)
        plt.xlabel(xlabel)
        plt.ylabel("density")
        plt.savefig(filename)
        plt.close()

    plot_hist(ditau_mass_list, 40, "ditau mass (GeV)", "ditau_mass.png")
    plot_hist(dijet_mass_list, 40, "dijet mass (GeV)", "dijet_mass.png")
    plot_hist(delta_eta_jj_list, 40, "|delta eta| between jets", "delta_eta_jj.png")
    plot_hist(leading_tau_pt_list, 40, "leading tau pT (GeV)", "leading_tau_pt.png")
    plot_hist(subleading_tau_pt_list, 40, "subleading tau pT (GeV)", "subleading_tau_pt.png")
    plot_hist(leading_jet_pt_list, 40, "leading jet pT (GeV)", "leading_jet_pt.png")
    plot_hist(subleading_jet_pt_list, 40, "subleading jet pT (GeV)", "subleading_jet_pt.png")
    plot_hist(delta_r_tt_list, 40, "delta R between taus", "delta_r_tt.png")
    plot_hist(delta_phi_tt_jj_list, 40, "delta phi between ditau and dijet", "delta_phi_tt_jj.png")
    plot_hist(centrality_tt_jj_list, 40, "centrality of ditau relative to jets", "centrality_tt_jj.png")

if __name__ == "__main__":
    main()
```
